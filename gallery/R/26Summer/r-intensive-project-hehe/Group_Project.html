#### Visually appealing heatmap

Hate the "0" data around the original data?\
Fill up the 0s around the original data and get yourselves a more appealing\
heatmap 🤗

```{r}
#| code-fold: false
#| echo: false

# Visually appealing heatmap
bin_a <- 1
bin_h <- 2
age_bins    <- seq(15, 42, by = bin_a)        
height_bins <- seq(160, 206, by = bin_h)

heat <- players %>%
  filter(between(age, min(age_bins), max(age_bins) + bin_a),
         between(height_cm, min(height_bins), max(height_bins) + bin_h)) %>%
  mutate(
    age_bin    = floor(age / bin_a) * bin_a,
    height_bin = floor(height_cm / bin_h) * bin_h
  ) %>%
  count(positions, age_bin, height_bin, name = "n") %>%
  complete(
    positions,
    age_bin    = age_bins,
    height_bin = height_bins,
    fill = list(n = 0)
  )

lo <- 1                  # treat 0 as background
hi <- max(heat$n, na.rm=TRUE)
ggplot(heat, aes(age_bin, height_bin, fill = n)) +
  geom_tile() +
  facet_wrap(~positions, ncol = 2) +
  scale_fill_viridis_c(
    trans = "sqrt",
    name  = "Count",
    limits = c(lo, hi),
    oob = scales::squish
  ) +
  scale_x_continuous(limits = c(15, 42), expand = c(0, 0)) +
  scale_y_continuous(limits = c(160, 206), expand = c(0, 0)) +
  theme_classic() +
  theme(panel.grid = element_blank())
```
